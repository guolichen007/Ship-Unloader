cmake_minimum_required(VERSION 3.16)
project(ship_perception VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
set(M0_EIGEN_VERSION "${EIGEN3_VERSION_STRING}")
get_target_property(M0_EIGEN_INCLUDE_DIRS Eigen3::Eigen INTERFACE_INCLUDE_DIRECTORIES)
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
include(cmake/SmallGicp.cmake)
set(M0_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/m0.json" CACHE FILEPATH "Versioned M0 config")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${M0_CONFIG}"
  "${CMAKE_CURRENT_SOURCE_DIR}/config/m0.schema.json" "${CMAKE_CURRENT_SOURCE_DIR}/tools/configure.py")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE M0_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE git_result)
if(NOT git_result EQUAL 0)
  set(M0_GIT_SHA UNCOMMITTED)
endif()
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/generated/ship_perception)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/configure.py
  ${M0_CONFIG} ${CMAKE_CURRENT_BINARY_DIR}/generated/ship_perception/config.hpp ${M0_GIT_SHA}
  RESULT_VARIABLE config_result)
if(NOT config_result EQUAL 0)
  message(FATAL_ERROR "Invalid M0 configuration")
endif()
set(V14_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/v14.json" CACHE FILEPATH "V1.4 配置")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${V14_CONFIG}"
  "${CMAKE_CURRENT_SOURCE_DIR}/config/v14.schema.json" "${CMAKE_CURRENT_SOURCE_DIR}/tools/configure_v14.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/configure_v14.py"
  "${V14_CONFIG}" "${CMAKE_CURRENT_BINARY_DIR}/generated/ship_perception/v14_config.hpp" RESULT_VARIABLE v14_result)
if(NOT v14_result EQUAL 0)
  message(FATAL_ERROR "V1.4 配置校验失败")
endif()
include(CTest)
add_library(ship_core src/replay/replay.cpp)
target_include_directories(ship_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(ship_core PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(ship_core PUBLIC Eigen3::Eigen)
add_library(ship_registration src/registration/quality.cpp)
target_link_libraries(ship_registration PUBLIC ship_core PRIVATE small_gicp_vendor)
option(M0_WITH_PCL "Build PCD I/O using system PCL (required on validation lane)" ON)
if(M0_WITH_PCL)
  # Ubuntu 20.04 PCL 1.10 may expose only versioned pkg-config modules.
  # Use the package's CMake config, including its transitive I/O dependencies.
  # The project enables C as well as CXX for PCL/VTK's MPI::MPI_C dependency.
  find_package(PCL 1.10 REQUIRED COMPONENTS common io)
  add_library(ship_pcd src/sensor/pcd.cpp)
  target_include_directories(ship_pcd SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
  target_compile_options(ship_pcd PRIVATE ${PCL_DEFINITIONS})
  target_link_libraries(ship_pcd PUBLIC ship_core PRIVATE ${PCL_LIBRARIES})
else()
  set(PCL_VERSION "DISABLED_DEVELOPER_ONLY")
  add_library(ship_pcd src/sensor/pcd_stub.cpp)
  target_link_libraries(ship_pcd PUBLIC ship_core)
endif()
# Record the dependencies actually selected by CMake, not unrelated .pc files
# elsewhere on the host. Plain key/value format preserves paths without JSON escaping.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build_dependencies.txt"
  "CMAKE_COMMAND=${CMAKE_COMMAND}\nCMAKE_VERSION=${CMAKE_VERSION}\n"
  "CXX_COMPILER=${CMAKE_CXX_COMPILER}\nCXX_COMPILER_VERSION=${CMAKE_CXX_COMPILER_VERSION}\n"
  "EIGEN_VERSION=${M0_EIGEN_VERSION}\nEIGEN_CONFIG_DIR=${Eigen3_DIR}\nEIGEN_INCLUDE_DIRS=${M0_EIGEN_INCLUDE_DIRS}\n"
  "PCL_VERSION=${PCL_VERSION}\nPCL_CONFIG_DIR=${PCL_DIR}\nPCL_INCLUDE_DIRS=${PCL_INCLUDE_DIRS}\n"
  "SMALL_GICP_VERSION=1.0.0\nSMALL_GICP_COMMIT=${SMALL_GICP_COMMIT}\nSMALL_GICP_THREADS=1\n")
add_executable(synthetic_replay tools/replay_main.cpp)
target_link_libraries(synthetic_replay PRIVATE ship_pcd)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ship_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
function(add_gate gate folder)
  add_executable(${gate} tests/${folder}/main.cpp)
  target_link_libraries(${gate} PRIVATE ship_core)
  target_include_directories(${gate} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${gate} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/artifacts/${gate})
  add_test(NAME ${gate} COMMAND ${gate})
  set_tests_properties(${gate} PROPERTIES
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/artifacts/${gate} TIMEOUT 120)
endfunction()
