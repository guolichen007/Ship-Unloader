cmake_minimum_required(VERSION 3.16)
project(ship_perception VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
set(M0_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/m0.json" CACHE FILEPATH "Versioned M0 config")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${M0_CONFIG}")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE M0_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE git_result)
if(NOT git_result EQUAL 0)
  set(M0_GIT_SHA UNCOMMITTED)
endif()
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/generated/ship_perception)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/configure.py
  ${M0_CONFIG} ${CMAKE_CURRENT_BINARY_DIR}/generated/ship_perception/config.hpp ${M0_GIT_SHA}
  RESULT_VARIABLE config_result)
if(NOT config_result EQUAL 0)
  message(FATAL_ERROR "Invalid M0 configuration")
endif()
include(CTest)
add_library(ship_core src/replay/replay.cpp)
target_include_directories(ship_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(ship_core PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(ship_core PUBLIC Eigen3::Eigen)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ship_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
function(add_gate gate folder)
  add_executable(${gate} tests/${folder}/main.cpp)
  target_link_libraries(${gate} PRIVATE ship_core)
  target_include_directories(${gate} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${gate} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/artifacts/${gate})
  add_test(NAME ${gate} COMMAND ${gate})
  set_tests_properties(${gate} PROPERTIES
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/artifacts/${gate} TIMEOUT 120)
endfunction()
if(BUILD_TESTING)
  add_gate(g1 g1_transform)
  add_gate(g2 g2_timestamp)
  add_gate(g3 g3_deskew)
  add_gate(g4 g4_dual_lidar)
  add_gate(g5 g5_known_se3)
  add_gate(g6 g6_float_precision)
  add_gate(g7 g7_latency)
  set_tests_properties(g3 g4 g5 PROPERTIES DEPENDS replay_selfcheck)
  add_executable(replay_selfcheck tests/replay_selfcheck.cpp)
  target_link_libraries(replay_selfcheck PRIVATE ship_core)
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/artifacts/replay_selfcheck)
  add_test(NAME replay_selfcheck COMMAND replay_selfcheck)
  set_tests_properties(replay_selfcheck PROPERTIES
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/artifacts/replay_selfcheck)
endif()
