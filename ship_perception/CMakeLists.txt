cmake_minimum_required(VERSION 3.16)
project(ship_perception VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
include(CTest)
add_library(ship_core INTERFACE)
target_include_directories(ship_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(ship_core INTERFACE Eigen3::Eigen)
